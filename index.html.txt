    <script>
        function generateVCF() {
            const input = document.getElementById('contactInput').value.trim();
            const commonPrefix = document.getElementById('prefix').value.trim();
            const filename = document.getElementById('filename').value.trim() || 'contactos';
            
            if (!input) {
                alert("Por favor, introduce al menos un número.");
                return;
            }

            const lines = input.split('\n');
            let vCardContent = "";
            let count = 0;

            lines.forEach((line, index) => {
                let name, phone;
                
                if (line.includes(',')) {
                    const parts = line.split(',');
                    name = parts[0].trim();
                    phone = parts[1].trim().replace(/\s+/g, '');
                } else {
                    phone = line.trim().replace(/\s+/g, '');
                    name = `${commonPrefix} ${index + 1}`.trim();
                }

                if (phone.length > 5) { // Filtro básico de número válido
                    vCardContent += "BEGIN:VCARD\n";
                    vCardContent += "VERSION:3.0\n";
                    vCardContent += `FN:${name}\n`;
                    vCardContent += `TEL;TYPE=CELL:${phone}\n`;
                    vCardContent += "END:VCARD\n";
                    count++;
                }
            });

            if (count === 0) {
                alert("No se detectaron números válidos.");
                return;
            }

            const blob = new Blob([vCardContent], { type: "text/vcard" });
            const url = window.URL.createObjectURL(blob);
            const a = document.createElement("a");
            a.href = url;
            a.download = `${filename}.vcf`;
            a.click();
            window.URL.revokeObjectURL(url);

            const statusDiv = document.getElementById('status');
            const statusText = document.getElementById('statusText');
            statusDiv.classList.remove('hidden');
            statusText.innerText = `¡Éxito! Se han procesado ${count} contactos.`;
        }
    </script>
